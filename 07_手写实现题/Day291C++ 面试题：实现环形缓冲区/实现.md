## 2. 实现

下面采用 `size_` 区分空和满，因此可以使用全部 `capacity` 个槽位。

```cpp
#include <cstddef>
#include <optional>
#include <stdexcept>
#include <utility>
#include <vector>

template <typename T>
class RingBuffer {
public:
    // 创建固定容量的缓冲区。容量在对象生命周期内保持不变，
    // 因此 push/pop 过程中不会触发底层数组扩容。
    explicit RingBuffer(size_t capacity) : data_(capacity) {
        // 容量为 0 时无法进行取模，也没有可用槽位，因此直接拒绝。
        if (capacity == 0) {
            throw std::invalid_argument("capacity must be greater than zero");
        }
    }

    // 将元素写入队尾。按值接收可以同时支持左值拷贝和右值移动。
    bool push(T value) {
        // 固定容量队列已满时不覆盖旧数据，由调用者决定重试或丢弃。
        if (full()) {
            return false;
        }

        // tail_ 始终指向下一个可写槽位。
        // optional 原本为空，赋值后该槽位开始保存一个有效对象。
        data_[tail_] = std::move(value);

        // 写入成功后将队尾移动一格；到达数组末尾时绕回下标 0。
        tail_ = next(tail_);
        ++size_;
        return true;
    }

    // 覆盖式入队：队列未满时正常追加，队列已满时覆盖最旧元素。
    // 适用于日志、音视频帧、传感器采样等“优先保留最新数据”的场景。
    void push_overwrite(T value) {
        const bool was_full = full();

        // 队列满时 tail_ == head_，因此这里覆盖的正是最旧元素。
        // optional 的移动赋值会正确释放槽位中旧对象持有的资源。
        data_[tail_] = std::move(value);

        // 无论是否已满，下一个写入位置都要向前移动一格。
        tail_ = next(tail_);

        if (was_full) {
            // 最旧元素已经被覆盖，head_ 必须移向新的最旧元素。
            // 队列中仍然有 capacity 个元素，所以 size_ 保持不变。
            head_ = next(head_);
        } else {
            // 队列原来未满，本次写入使有效元素数量增加一个。
            ++size_;
        }
    }

    // 取出队头元素，并通过 value 返回。
    bool pop(T& value) {
        // 空队列没有元素可读，不能解引用 data_[head_]。
        if (empty()) {
            return false;
        }

        // head_ 始终指向当前最早入队的元素。
        // 使用移动赋值可以减少 T 内部资源的复制开销。
        value = std::move(*data_[head_]);

        // 清空 optional，使槽位中的 T 立即析构并重新变为可写状态。
        data_[head_].reset();

        // 读取成功后移动队头，并减少有效元素数量。
        head_ = next(head_);
        --size_;
        return true;
    }

    // 返回队头元素的引用，但不执行出队。
    T& front() {
        if (empty()) {
            throw std::out_of_range("ring buffer is empty");
        }
        return *data_[head_];
    }

    // const 对象只能取得只读引用，避免通过 front() 修改元素。
    const T& front() const {
        if (empty()) {
            throw std::out_of_range("ring buffer is empty");
        }
        return *data_[head_];
    }

    // 通过 size_ 区分空和满，因此 capacity 个槽位都可以使用。
    bool empty() const noexcept { return size_ == 0; }
    bool full() const noexcept { return size_ == data_.size(); }
    size_t size() const noexcept { return size_; }
    size_t capacity() const noexcept { return data_.size(); }

private:
    size_t next(size_t index) const noexcept {
        // 取模让物理数组的首尾在逻辑上连接成一个环。
        return (index + 1) % data_.size();
    }

    // optional 让槽位可以处于“尚未构造 T”的空状态，
    // 因而不要求 T 必须提供默认构造函数。
    std::vector<std::optional<T>> data_;
    size_t head_ = 0; // 当前队头，也就是下一次 pop 的读取位置
    size_t tail_ = 0; // 当前队尾之后，也就是下一次 push 的写入位置
    size_t size_ = 0; // 当前有效元素数，同时用于区分 head_ == tail_ 时的空和满
};
```
