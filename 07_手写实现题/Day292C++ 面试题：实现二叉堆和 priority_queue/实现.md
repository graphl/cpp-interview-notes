```
#include <functional>
#include <cstddef>
#include <stdexcept>
#include <utility>
#include <vector>

template <typename T, typename Compare = std::less<T>>
class PriorityQueue {
public:
    PriorityQueue() = default;

    explicit PriorityQueue(std::vector<T> values)
        : heap_(std::move(values)) {
        heapify();
    }

    void push(T value) {
        heap_.push_back(std::move(value));
        sift_up(heap_.size() - 1);
    }

    void pop() {
        if (heap_.empty()) {
            throw std::out_of_range("priority queue is empty");
        }
        std::swap(heap_.front(), heap_.back());
        heap_.pop_back();
        if (!heap_.empty()) {
            sift_down(0);
        }
    }

    const T& top() const {
        if (heap_.empty()) {
            throw std::out_of_range("priority queue is empty");
        }
        return heap_.front();
    }

    bool empty() const noexcept { return heap_.empty(); }
    size_t size() const noexcept { return heap_.size(); }

private:
    // comp_(a, b) 为 true，表示 a 的优先级低于 b。
    bool lower(size_t a, size_t b) const {
        return comp_(heap_[a], heap_[b]);
    }

    void sift_up(size_t child) {
        while (child > 0) {
            size_t parent = (child - 1) / 2;
            if (!lower(parent, child)) {
                break;
            }
            std::swap(heap_[parent], heap_[child]);
            child = parent;
        }
    }

    void sift_down(size_t parent) {
        const size_t n = heap_.size();
        while (true) {
            size_t best = parent;
            size_t left = parent * 2 + 1;
            size_t right = left + 1;

            if (left < n && lower(best, left)) best = left;
            if (right < n && lower(best, right)) best = right;
            if (best == parent) break;

            std::swap(heap_[parent], heap_[best]);
            parent = best;
        }
    }

    void heapify() {
        if (heap_.size() < 2) return;
        for (size_t i = heap_.size() / 2; i > 0; --i) {
            sift_down(i - 1);
        }
    }

    std::vector<T> heap_;
    Compare comp_{};
};
```

默认 `std::less<T>` 得到大根堆；使用 `std::greater<T>` 得到小根堆。
