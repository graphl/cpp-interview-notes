# 2. 拉链法实现

   

```
#include <functional>
#include <list>
#include <utility>
#include <vector>

template <typename K, typename V>
class HashMap {
public:
    // 预先创建 bucket_count 个桶；每个桶内部是一条链表。
    // explicit 防止整数被隐式转换成 HashMap 对象。
    explicit HashMap(size_t bucket_count = 16)
        : buckets_(bucket_count), size_(0) {}

    void put(const K& key, const V& value) {
        // 负载因子过高时，冲突概率和链表查询长度都会增加，
        // 因此将桶数量扩大一倍，并重新计算所有元素所在的桶。
        if (load_factor() > 0.75) {
            rehash(buckets_.size() * 2);
        }

        // 哈希值映射为桶下标，只在对应链表中查找。
        size_t idx = index(key);
        for (auto& kv : buckets_[idx]) {
            if (kv.first == key) {
                // key 已存在时只更新 value，元素总数不变。
                kv.second = value;
                return;
            }
        }

        // key 不存在，将新键值对挂到冲突链表末尾。
        buckets_[idx].push_back({key, value});
        ++size_;
    }

    bool get(const K& key, V& value) const {
        size_t idx = index(key);
        for (const auto& kv : buckets_[idx]) {
            if (kv.first == key) {
                // 通过输出参数返回查询结果；返回值表示是否找到。
                value = kv.second;
                return true;
            }
        }
        return false;
    }

    bool erase(const K& key) {
        size_t idx = index(key);
        // 使用引用直接操作 buckets_ 中的链表，避免复制整个桶。
        auto& bucket = buckets_[idx];

        for (auto it = bucket.begin(); it != bucket.end(); ++it) {
            if (it->first == key) {
                // std::list 擦除当前节点不会搬移其他节点。
                bucket.erase(it);
                --size_;
                return true;
            }
        }

        return false;
    }

    size_t size() const {
        return size_;
    }

private:
    // 拉链法：一个桶保存一条键值对链表，哈希冲突的元素位于同一链表中。
    using Bucket = std::list<std::pair<K, V>>;

    size_t index(const K& key) const {
        // std::hash<K> 生成哈希值，取模后落到有效桶下标范围内。
        return std::hash<K>{}(key) % buckets_.size();
    }

    double load_factor() const {
        // 显式转换为 double，避免整数除法丢失小数部分。
        return static_cast<double>(size_) / buckets_.size();
    }

    void rehash(size_t new_bucket_count) {
        // 先接管旧桶。桶数变化后，原来的下标不再有效，
        // 所有键值对都必须依据新的桶数量重新计算位置。
        std::vector<Bucket> old_buckets = std::move(buckets_);
        buckets_.clear();
        buckets_.resize(new_bucket_count);
        size_ = 0;

        // 复用 put 完成重新插入，同时恢复 size_ 计数。
        for (const auto& bucket : old_buckets) {
            for (const auto& kv : bucket) {
                put(kv.first, kv.second);
            }
        }
    }

    // 桶数组：下标由 key 的哈希值决定。
    std::vector<Bucket> buckets_;
    // 实际存储的键值对数量，不是桶数量。
    size_t size_;
};
```
