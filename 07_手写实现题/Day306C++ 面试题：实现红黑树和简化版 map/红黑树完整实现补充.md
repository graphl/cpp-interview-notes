# C++ 面试题：红黑树完整插入与删除补充

## 1. 实现范围

下面实现一个不允许重复 key 的教学版红黑树，补齐：

1. 左旋与右旋。
2. 插入的左右镜像修复。
3. 查找和删除。
4. 删除后的双黑修复。
5. 红黑树不变量验证。

代码重点是树结构和颜色修复，不包含完整标准库 `map` 的迭代器、分配器、异常保证和异构查找接口。

## 2. 完整教学实现

```cpp
#include <algorithm>
#include <cstddef>
#include <functional>
#include <utility>

template <typename Key, typename Value,
          typename Compare = std::less<Key>>
class RBTree {
    enum class Color { Red, Black };

    struct Node {
        Key key;
        Value value;
        Color color = Color::Red;
        Node* parent = nullptr;
        Node* left = nullptr;
        Node* right = nullptr;
    };

public:
    RBTree() = default;
    RBTree(const RBTree&) = delete;
    RBTree& operator=(const RBTree&) = delete;

    ~RBTree() {
        clear(root_);
    }

    Value* find(const Key& key) noexcept {
        Node* node = find_node(key);
        return node ? &node->value : nullptr;
    }

    const Value* find(const Key& key) const noexcept {
        Node* node = find_node(key);
        return node ? &node->value : nullptr;
    }

    bool insert(Key key, Value value) {
        Node* parent = nullptr;
        Node* current = root_;

        while (current) {
            parent = current;
            if (compare_(key, current->key)) {
                current = current->left;
            } else if (compare_(current->key, key)) {
                current = current->right;
            } else {
                return false;
            }
        }

        Node* node = new Node{
            std::move(key), std::move(value), Color::Red,
            parent, nullptr, nullptr
        };

        if (!parent) {
            root_ = node;
        } else if (compare_(node->key, parent->key)) {
            parent->left = node;
        } else {
            parent->right = node;
        }

        fix_after_insert(node);
        ++size_;
        return true;
    }

    bool erase(const Key& key) {
        Node* target = find_node(key);
        if (!target) {
            return false;
        }

        Node* removed = target;
        Color removed_color = removed->color;
        Node* child = nullptr;
        Node* child_parent = nullptr;

        if (!target->left) {
            child = target->right;
            child_parent = target->parent;
            transplant(target, target->right);
        } else if (!target->right) {
            child = target->left;
            child_parent = target->parent;
            transplant(target, target->left);
        } else {
            removed = minimum(target->right);
            removed_color = removed->color;
            child = removed->right;

            if (removed->parent == target) {
                child_parent = removed;
                if (child) {
                    child->parent = removed;
                }
            } else {
                child_parent = removed->parent;
                transplant(removed, removed->right);
                removed->right = target->right;
                removed->right->parent = removed;
            }

            transplant(target, removed);
            removed->left = target->left;
            removed->left->parent = removed;
            removed->color = target->color;
        }

        delete target;
        --size_;

        if (removed_color == Color::Black) {
            fix_after_erase(child, child_parent);
        }
        return true;
    }

    std::size_t size() const noexcept {
        return size_;
    }

    bool validate() const {
        if (!root_) {
            return size_ == 0;
        }
        if (root_->color != Color::Black || root_->parent) {
            return false;
        }

        std::size_t node_count = 0;
        int black_height = validate_node(
            root_, nullptr, nullptr, node_count);
        return black_height > 0 && node_count == size_;
    }

private:
    static Color color_of(const Node* node) noexcept {
        return node ? node->color : Color::Black;
    }

    static void set_color(Node* node, Color color) noexcept {
        if (node) {
            node->color = color;
        }
    }

    Node* find_node(const Key& key) const noexcept {
        Node* current = root_;
        while (current) {
            if (compare_(key, current->key)) {
                current = current->left;
            } else if (compare_(current->key, key)) {
                current = current->right;
            } else {
                return current;
            }
        }
        return nullptr;
    }

    static Node* minimum(Node* node) noexcept {
        while (node->left) {
            node = node->left;
        }
        return node;
    }

    void rotate_left(Node* node) noexcept {
        Node* pivot = node->right;
        node->right = pivot->left;
        if (pivot->left) {
            pivot->left->parent = node;
        }

        pivot->parent = node->parent;
        if (!node->parent) {
            root_ = pivot;
        } else if (node == node->parent->left) {
            node->parent->left = pivot;
        } else {
            node->parent->right = pivot;
        }

        pivot->left = node;
        node->parent = pivot;
    }

    void rotate_right(Node* node) noexcept {
        Node* pivot = node->left;
        node->left = pivot->right;
        if (pivot->right) {
            pivot->right->parent = node;
        }

        pivot->parent = node->parent;
        if (!node->parent) {
            root_ = pivot;
        } else if (node == node->parent->right) {
            node->parent->right = pivot;
        } else {
            node->parent->left = pivot;
        }

        pivot->right = node;
        node->parent = pivot;
    }

    void fix_after_insert(Node* node) noexcept {
        while (node->parent &&
               node->parent->color == Color::Red) {
            Node* parent = node->parent;
            Node* grand = parent->parent;

            if (parent == grand->left) {
                Node* uncle = grand->right;
                if (color_of(uncle) == Color::Red) {
                    set_color(parent, Color::Black);
                    set_color(uncle, Color::Black);
                    set_color(grand, Color::Red);
                    node = grand;
                } else {
                    if (node == parent->right) {
                        node = parent;
                        rotate_left(node);
                        parent = node->parent;
                        grand = parent->parent;
                    }
                    set_color(parent, Color::Black);
                    set_color(grand, Color::Red);
                    rotate_right(grand);
                }
            } else {
                Node* uncle = grand->left;
                if (color_of(uncle) == Color::Red) {
                    set_color(parent, Color::Black);
                    set_color(uncle, Color::Black);
                    set_color(grand, Color::Red);
                    node = grand;
                } else {
                    if (node == parent->left) {
                        node = parent;
                        rotate_right(node);
                        parent = node->parent;
                        grand = parent->parent;
                    }
                    set_color(parent, Color::Black);
                    set_color(grand, Color::Red);
                    rotate_left(grand);
                }
            }
        }
        set_color(root_, Color::Black);
    }

    void transplant(Node* old_node, Node* new_node) noexcept {
        if (!old_node->parent) {
            root_ = new_node;
        } else if (old_node == old_node->parent->left) {
            old_node->parent->left = new_node;
        } else {
            old_node->parent->right = new_node;
        }
        if (new_node) {
            new_node->parent = old_node->parent;
        }
    }

    void fix_after_erase(Node* node, Node* parent) noexcept {
        while (node != root_ &&
               color_of(node) == Color::Black) {
            if (!parent) {
                break;
            }

            if (node == parent->left) {
                Node* sibling = parent->right;

                if (color_of(sibling) == Color::Red) {
                    set_color(sibling, Color::Black);
                    set_color(parent, Color::Red);
                    rotate_left(parent);
                    sibling = parent->right;
                }

                if (color_of(sibling ? sibling->left : nullptr) ==
                        Color::Black &&
                    color_of(sibling ? sibling->right : nullptr) ==
                        Color::Black) {
                    set_color(sibling, Color::Red);
                    node = parent;
                    parent = node->parent;
                } else {
                    if (color_of(
                            sibling ? sibling->right : nullptr) ==
                        Color::Black) {
                        set_color(
                            sibling ? sibling->left : nullptr,
                            Color::Black);
                        set_color(sibling, Color::Red);
                        rotate_right(sibling);
                        sibling = parent->right;
                    }

                    set_color(sibling, color_of(parent));
                    set_color(parent, Color::Black);
                    set_color(
                        sibling ? sibling->right : nullptr,
                        Color::Black);
                    rotate_left(parent);
                    node = root_;
                    parent = nullptr;
                }
            } else {
                Node* sibling = parent->left;

                if (color_of(sibling) == Color::Red) {
                    set_color(sibling, Color::Black);
                    set_color(parent, Color::Red);
                    rotate_right(parent);
                    sibling = parent->left;
                }

                if (color_of(sibling ? sibling->right : nullptr) ==
                        Color::Black &&
                    color_of(sibling ? sibling->left : nullptr) ==
                        Color::Black) {
                    set_color(sibling, Color::Red);
                    node = parent;
                    parent = node->parent;
                } else {
                    if (color_of(
                            sibling ? sibling->left : nullptr) ==
                        Color::Black) {
                        set_color(
                            sibling ? sibling->right : nullptr,
                            Color::Black);
                        set_color(sibling, Color::Red);
                        rotate_left(sibling);
                        sibling = parent->left;
                    }

                    set_color(sibling, color_of(parent));
                    set_color(parent, Color::Black);
                    set_color(
                        sibling ? sibling->left : nullptr,
                        Color::Black);
                    rotate_right(parent);
                    node = root_;
                    parent = nullptr;
                }
            }
        }
        set_color(node, Color::Black);
    }

    int validate_node(const Node* node,
                      const Key* lower,
                      const Key* upper,
                      std::size_t& count) const {
        if (!node) {
            return 1;
        }

        if ((lower && !compare_(*lower, node->key)) ||
            (upper && !compare_(node->key, *upper))) {
            return -1;
        }
        if (node->left && node->left->parent != node) {
            return -1;
        }
        if (node->right && node->right->parent != node) {
            return -1;
        }
        if (node->color == Color::Red &&
            (color_of(node->left) == Color::Red ||
             color_of(node->right) == Color::Red)) {
            return -1;
        }

        ++count;
        int left_height = validate_node(
            node->left, lower, &node->key, count);
        int right_height = validate_node(
            node->right, &node->key, upper, count);

        if (left_height < 0 || right_height < 0 ||
            left_height != right_height) {
            return -1;
        }
        return left_height +
               (node->color == Color::Black ? 1 : 0);
    }

    static void clear(Node* node) noexcept {
        if (!node) {
            return;
        }
        clear(node->left);
        clear(node->right);
        delete node;
    }

    Node* root_ = nullptr;
    std::size_t size_ = 0;
    Compare compare_;
};
```

## 3. 删除为什么最难

删除红节点不会改变黑高；删除黑节点会让一侧少一个黑节点，因此需要把额外的“黑”向上转移，或者通过兄弟节点的颜色、孩子颜色和旋转消除它。

实现允许 `node == nullptr`，所以修复函数必须额外携带 `parent`。如果使用共享的黑色哨兵节点，父指针处理会更统一，但会增加类的状态。

## 4. 复杂度和边界

1. 查找、插入和删除都是 `O(log n)`。
2. 旋转是 `O(1)`，且保持中序 key 顺序。
3. 插入重复 key 返回 `false`。
4. 验证函数同时检查 BST 顺序、父指针、红红冲突、黑高和节点数量。
5. `new Node` 成功后才修改树，因此节点分配失败不会改变原树。

## 5. 面试口述版

红黑树插入主要修复红父红子冲突，分为叔叔为红时变色、折线时先旋转父节点、直线时旋转祖父节点。删除黑节点可能破坏黑高，需要根据兄弟是否为红、兄弟孩子的颜色进行变色和旋转。左右分支完全镜像，写完后应使用不变量验证函数检查根为黑、红节点无红孩子、所有根叶路径黑高相同。
