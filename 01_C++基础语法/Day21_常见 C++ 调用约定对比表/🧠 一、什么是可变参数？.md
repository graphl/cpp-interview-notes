# 🧠 一、什么是可变参数？



```
#include <cstdarg>
#include <iostream>

void printNumbers(int count, ...) {
    va_list args;
    va_start(args, count);  // 初始化可变参数

    for (int i = 0; i < count; ++i) {
        int val = va_arg(args, int);  // 获取一个 int 参数
        std::cout << val << " ";
    }
    
    va_end(args);  // 结束
    std::cout << std::endl;
}

/// VS中设置成__stdcall能编译成功是因为编译器会默默地将它转换为 __cdecl 调用约定，以保证可变参数的正确处理和栈清理方式 
#include <iostream>
template<typename T>
void print(T val) {
    std::cout << val << " ";
}
template<typename T, typename... Args>
void print(T first, Args... args) {
    std::cout << first << " ";
    print(args...);  // 递归展开参数包
}
```
