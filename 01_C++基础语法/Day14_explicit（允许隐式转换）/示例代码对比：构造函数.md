# 示例代码对比：构造函数



```
// 没有 `explicit`
class A {
public:
    A(int x) {}  // 可隐式转换
};

void func(A a) {}

int main() {
    func(10);  // ✅ 合法，10 被隐式转换为 A
}

// 使用 explicit 后
class A {
public:
    explicit A(int x) {}  // 禁止隐式转换
};

void func(A a) {}

int main() {
    func(10);     // ❌ 编译错误
    func(A(10));  // ✅ 正确：显示构造
}

```
